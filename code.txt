<script>

const promptBox =
document.getElementById("prompt");

const suggestions = [
"Cyberpunk city at sunset",
"Anime warrior girl",
"Luxury sports car",
"Fantasy castle",
"Dragon flying over mountains",
"Robot samurai",
"Space station on Mars",
"Underwater kingdom"
];

function setPrompt(text){
promptBox.value=text;
}

function randomPrompt(){
promptBox.value=
suggestions[Math.floor(Math.random()*suggestions.length)];
}

function copyPrompt(){
navigator.clipboard.writeText(promptBox.value);
alert("Prompt copied");
}

function clearGallery(){
document.getElementById("gallery").innerHTML="";
}

promptBox.addEventListener("keypress",e=>{
if(e.key==="Enter"){
generateImage();
}
});

function generateImage(){

const prompt =
promptBox.value.trim();

if(!prompt){
alert("Enter a prompt");
return;
}

const style =
document.getElementById("style").value;

document.getElementById("loader").style.display="block";

const finalPrompt =
style + " " + prompt;

const url =
`https://image.pollinations.ai/prompt/${encodeURIComponent(finalPrompt)}?seed=${Math.random()}`;

const img = new Image();

img.onload = ()=>{

document.getElementById("loader").style.display="none";

const card =
document.createElement("div");

card.className="card";

card.innerHTML = `
<img src="${url}" onclick="openModal('${url}')">

<button class="download-btn"
onclick="window.open('${url}')">
Download
</button>

<div class="prompt-text">
${finalPrompt}
</div>
`;

document.getElementById("gallery")
.prepend(card);

};

img.src=url;

}

function openModal(src){
document.getElementById("modal").style.display="flex";
document.getElementById("modalImg").src=src;
}

function closeModal(){
document.getElementById("modal").style.display="none";
}

</script>
